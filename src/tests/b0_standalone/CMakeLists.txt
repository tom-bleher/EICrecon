cmake_minimum_required(VERSION 3.16)
project(b0_standalone_tests LANGUAGES CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_path(EIGEN3_INCLUDE_DIR Eigen/Core PATH_SUFFIXES eigen3)
file(GLOB B0_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp")
foreach(source IN LISTS B0_TEST_SOURCES)
  get_filename_component(name "${source}" NAME_WE)
  add_executable("${name}" "${source}")
  if(name STREQUAL "test_telescope_math")
    if(NOT EIGEN3_INCLUDE_DIR)
      message(FATAL_ERROR "Telescope tests require Eigen3; set EIGEN3_INCLUDE_DIR")
    endif()
    target_sources("${name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../algorithms/tracking/B0TelescopeMath.cc")
  endif()
  target_include_directories("${name}" PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../algorithms/tracking")
  if(EIGEN3_INCLUDE_DIR)
    target_include_directories("${name}" SYSTEM PRIVATE "${EIGEN3_INCLUDE_DIR}")
  endif()
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options("${name}" PRIVATE -Wall -Wextra -Werror -pedantic)
  endif()
  add_test(NAME "${name}" COMMAND "${name}")
endforeach()
