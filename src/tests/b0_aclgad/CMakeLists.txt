cmake_minimum_required(VERSION 3.16)
project(B0ACLGADStandalone LANGUAGES CXX)
enable_testing()
add_executable(test_response test_response.cpp)
target_compile_features(test_response PRIVATE cxx_std_17)
target_include_directories(test_response PRIVATE ../../algorithms/digi)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_response PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME b0_aclgad_response COMMAND test_response)
add_executable(test_calibration test_calibration.cpp)
target_compile_features(test_calibration PRIVATE cxx_std_17)
target_include_directories(test_calibration PRIVATE ../../algorithms/digi)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_calibration PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME b0_aclgad_calibration COMMAND test_calibration ${CMAKE_CURRENT_SOURCE_DIR}/../../../docs/b0-aclgad/experimental.cfg)
add_executable(test_truth test_truth.cpp)
target_compile_features(test_truth PRIVATE cxx_std_17)
target_include_directories(test_truth PRIVATE ../../algorithms/tracking)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(test_truth PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME b0_aclgad_truth_weights COMMAND test_truth)
